package variant3_multifunctionprinter;

// Опис предметної області: Офісна периферійна техніка та документообіг.
// Завдання коду: Керування процесами друку, сканування, степлювання та відправки факсів через єдину систему управління.

public class MultiFunctionPrinter {

    public interface IMultiFunctionDevice {
        void printDocument(String doc);
        void scanDocument();
        void sendFax(String number, String doc);
        void staplePages(int count);
        void duplexPrint(String doc);
    }

    public static class BasicLaserPrinter implements IMultiFunctionDevice {
        @Override public void printDocument(String doc) { System.out.println("Printing: " + doc); }
        @Override public void duplexPrint(String doc) { System.out.println("Duplex printing: " + doc); }

        @Override public void scanDocument() { throw new UnsupportedOperationException("Basic laser printer has no scanner unit."); }
        @Override public void sendFax(String number, String doc) { throw new UnsupportedOperationException("Basic printer cannot send faxes."); }
        @Override public void staplePages(int count) { throw new UnsupportedOperationException("Basic printer has no stapler module."); }
    }

    public static class DocumentScanner implements IMultiFunctionDevice {
        @Override public void scanDocument() { System.out.println("Scanning physical document..."); }

        @Override public void printDocument(String doc) { throw new UnsupportedOperationException("Scanner cannot print documents."); }
        @Override public void sendFax(String number, String doc) { throw new UnsupportedOperationException("Scanner cannot send faxes."); }
        @Override public void staplePages(int count) { throw new UnsupportedOperationException("Scanner cannot staple pages."); }
        @Override public void duplexPrint(String doc) { throw new UnsupportedOperationException("Scanner cannot print."); }
    }
}
